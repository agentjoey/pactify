**Goal:** `pactify mcp` — a stdio MCP server (one per agent session, single project via cwd/`PACT_DIR`, seat = `PACT_AGENT_ID`) exposing all pact verbs as tools, `pact://state` + `pact://log` as resources, and resource-updated notifications on log changes.

**Architecture:** New package `internal/mcp` wraps `internal/pact` verbs as MCP tool handlers (invariants fail-closed exactly like the CLI — the engine enforces them), reads STATE/log for resources, and reuses fsnotify to fire `ResourceUpdated` notifications. `cmd/pactify/cmd_mcp.go` runs it over `mcp.StdioTransport`. Tests use the SDK's in-memory transports.

**Tech Stack:** Go 1.26, `github.com/modelcontextprotocol/go-sdk/mcp` (official, new dep), existing `internal/{pact,paths,event,projection}`, fsnotify. Tests: `go test` (in-memory MCP client), `bats` (stdio smoke).

```go
package mcp

import (
	"context"
	"os"

	sdk "github.com/modelcontextprotocol/go-sdk/mcp"
	"github.com/agentjoey/pactify/internal/paths"
)

const (
	stateURI = "pact://state"
	logURI   = "pact://log"
)

func readFileResource(uri, path, mime string) sdk.ResourceHandler {
	return func(_ context.Context, _ *sdk.ReadResourceRequest) (*sdk.ReadResourceResult, error) {
		b, err := os.ReadFile(path)
		if err != nil && !os.IsNotExist(err) {
			return nil, err
		}
		return &sdk.ReadResourceResult{Contents: []*sdk.ResourceContents{{URI: uri, MIMEType: mime, Text: string(b)}}}, nil
	}
}

func registerResources(s *sdk.Server) {
	s.AddResource(&sdk.Resource{URI: stateURI, Name: "pact STATE.yml", MIMEType: "application/yaml",
		Description: "The rendered pact state (projection of the log; the log is authoritative)"},
		func(ctx context.Context, req *sdk.ReadResourceRequest) (*sdk.ReadResourceResult, error) {
			return readFileResource(stateURI, paths.State(), "application/yaml")(ctx, req)
		})
	s.AddResource(&sdk.Resource{URI: logURI, Name: "pact log.jsonl", MIMEType: "application/jsonl",
		Description: "The authoritative append-only pact event log"},
		func(ctx context.Context, req *sdk.ReadResourceRequest) (*sdk.ReadResourceResult, error) {
			return readFileResource(logURI, paths.Log(), "application/jsonl")(ctx, req)
		})
}
```
> Note: the handlers resolve `paths.State()/paths.Log()` at request time (not registration time) so a `PACT_DIR` change between tests is honored.

```go
import (
	"context"

	"github.com/fsnotify/fsnotify"
	"github.com/agentjoey/pactify/internal/paths"
	sdk "github.com/modelcontextprotocol/go-sdk/mcp"
)

// Watch starts an fsnotify watcher on the project's .pact/ dir and fires
// ResourceUpdated for pact://log and pact://state when log.jsonl changes.
// Returns a stop function.
func Watch(ctx context.Context, s *sdk.Server) (func(), error) {
	w, err := fsnotify.NewWatcher()
	if err != nil {
		return nil, err
	}
	if err := w.Add(paths.Dir()); err != nil {
		w.Close()
		return nil, err
	}
	go func() {
		for {
			select {
			case ev, ok := <-w.Events:
				if !ok {
					return
				}
				if ev.Op&(fsnotify.Write|fsnotify.Create) == 0 || ev.Name != paths.Log() {
					continue
				}
				_ = s.ResourceUpdated(ctx, &sdk.ResourceUpdatedNotificationParams{URI: logURI})
				_ = s.ResourceUpdated(ctx, &sdk.ResourceUpdatedNotificationParams{URI: stateURI})
			case <-ctx.Done():
				return
			case _, ok := <-w.Errors:
				if !ok {
					return
				}
			}
		}
	}()
	return func() { _ = w.Close() }, nil
}
```

```go
package main

import (
	"context"

	pactmcp "github.com/agentjoey/pactify/internal/mcp"
	sdk "github.com/modelcontextprotocol/go-sdk/mcp"
	"github.com/spf13/cobra"
)

func newMCPCmd() *cobra.Command {
	return &cobra.Command{Use: "mcp", Short: "run the stdio MCP server (tools+resources for this project)",
		RunE: func(c *cobra.Command, _ []string) error {
			ctx := context.Background()
			server := pactmcp.New()
			stop, err := pactmcp.Watch(ctx, server)
			if err == nil {
				defer stop()
			} // a missing .pact dir is not fatal: tools will fail-closed per call
			return server.Run(ctx, &sdk.StdioTransport{})
		}}
}
```
In `cmd/pactify/commands.go`, add `newMCPCmd()` to `root.AddCommand(...)`.
